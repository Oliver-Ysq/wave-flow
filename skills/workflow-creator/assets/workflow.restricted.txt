const meta = {
  name: "planned-workflow",
  description: "把目标拆解为独立任务，并行执行、用户确认后汇总结论。",
  phases: [{ title: "计划" }, { title: "执行" }, { title: "确认" }, { title: "汇总" }],
  exampleArgs: { objective: "比较三种方案并给出推荐" },
};

type Args = { objective: string };
type Plan = { tasks: string[] };

/**
 * 主控受限草案模板：不是模块，禁止 import/export；作者 API 由 Runtime 注入。
 * 可用的全局只有 agent、phase、parallel、pipeline、log、decisionGate、JSON、Error。
 * 护栏：agent() ≤ 20 次，phase() 切换 ≤ 50 次，decisionGate() ≤ 10 次。
 */
async function run(args: Args) {
  phase("计划");
  log("## 计划\n正在把目标拆解为互相独立的任务。");
  const plan = await agent(`把这个目标拆解为互相独立的任务清单：\n${args.objective}`, {
    id: "planner",
    cli: "codex",
    schema: {
      type: "object",
      properties: { tasks: { type: "array", items: { type: "string" }, minItems: 1 } },
      required: ["tasks"],
      additionalProperties: false,
    },
  }) as Plan | null;
  if (!plan) throw new Error("计划 Agent 未能交付结构化结果。");

  phase("执行");
  log(`## 执行\n并行执行 ${plan.tasks.length} 个任务。`);
  const raw = await parallel(
    plan.tasks.map((task, index) => async () => {
      log(`- 开始 ${task}`);
      const finding = await agent(`完成这项独立任务：\n${task}`, { id: `task-${index + 1}`, cli: "codex" });
      if (finding) log(`- 完成 ${task}`);
      return finding;
    }),
  );
  const findings = raw.filter((item) => item !== null);

  phase("确认");
  const decision = await decisionGate(
    "confirm-synthesis",
    `并行任务完成 ${findings.length}/${plan.tasks.length}。是否基于这些结果进入汇总？`,
    {
      type: "object",
      properties: { approved: { type: "boolean" }, note: { type: "string" } },
      required: ["approved"],
      additionalProperties: false,
    },
  );
  if (decision.approved !== true) throw new Error(`用户未批准进入汇总：${decision.note ?? "无说明"}`);

  phase("汇总");
  return agent(`原始目标：${args.objective}\n基于以下任务结果给出简明结论：\n${JSON.stringify(findings)}`, {
    id: "synthesizer",
    cli: "codex",
  });
}
